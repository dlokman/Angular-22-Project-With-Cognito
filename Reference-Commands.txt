To create a new User:

  aws cognito-idp admin-create-user `
    --user-pool-id us-east-1_UqVgmTUWo `
    --username dan.lokman@hotmail.com `
    --temporary-password "TempPass#12345" `
    --user-attributes "Name=email,Value=dan.lokman@hotmail.com" "Name=email_verified,Value=true" `
    --no-cli-pager

  aws cognito-idp admin-add-user-to-group `
    --user-pool-id us-east-1_UqVgmTUWo `
    --username dan.lokman@hotmail.com `
    --group-name admin `
    --no-cli-pager
